import io
import pandas as pd
from pypdf import PdfReader
import streamlit as st

st.set_page_config(
    page_title="PDF to Excel Converter", page_icon="📄", layout="centered"
)

st.title("📄 PDF to Excel 변환기 (2페이지 이후부터)")
st.write(
    "스캔된 PDF 파일을 업로드하시면, **2페이지부터 끝까지**의 내용을 추출하여"
    " 엑셀 파일로 변환해 드립니다."
)

# 파일 업로드
uploaded_file = st.file_uploader(
    "PDF 파일을 선택하세요.", type=["pdf"]
)

if uploaded_file is not None:
  try:
    # PDF 읽기
    reader = PdfReader(uploaded_file)
    total_pages = len(reader.pages)

    st.info(f"총 페이지 수: {total_pages}페이지")

    if total_pages < 2:
      st.warning(
          "업로드하신 PDF는 1페이지밖에 없습니다. 2페이지 이상인 파일을"
          " 업로드해주세요."
      )
    else:
      st.success(
          "2페이지부터 끝까지의 텍스트 추출을 준비 중입니다..."
      )

      # 2페이지부터 텍스트 추출
      extracted_data = []
      for index in range(1, total_pages):  # 0이 1페이지이므로 1부터 시작 (2페이지)
        page = reader.pages[index]
        text = page.extract_text()

        if text:
          # 줄 단위로 분리하여 데이터 구성
          lines = text.split("\n")
          for line_idx, line in enumerate(lines):
            if line.strip():
              extracted_data.append({
                  "페이지": index + 1,
                  "라인 번호": line_idx + 1,
                  "추출 내용": line.strip(),
              })
        else:
          extracted_data.append({
              "페이지": index + 1,
              "라인 번호": 1,
              "추출 내용": "(텍스트 없음 또는 이미지 스캔본)",
          })

      # DataFrame 변환
      df = pd.DataFrame(extracted_data)

      st.write("### 추출된 미리보기 (상위 10행)")
      st.dataframe(df.head(10))

      # 엑셀 파일 바이너리 생성
      output = io.BytesIO()
      with pd.ExcelWriter(output, engine="openpyxl") as writer:
        df.to_excel(writer, index=False, sheet_name="추출데이터")
      excel_data = output.getvalue()

      # 다운로드 버튼 제공
      st.download_button(
          label="📥 변환된 엑셀 파일 다운로드 (.xlsx)",
          data=excel_data,
          file_name="converted_from_page2.xlsx",
          mime=(
              "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
          ),
      )

  except Exception as e:
    st.error(f"파일을 처리하는 중 오류가 발생했습니다: {e}")
